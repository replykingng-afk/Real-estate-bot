# -------------------- FastAPI App --------------------
app = FastAPI(title="Gemini + Supabase API", description="Example integration")

# -------------------- Pydantic Models --------------------
class GenerateRequest(BaseModel):
    prompt: str
    temperature: Optional[float] = 0.7

class GenerateResponse(BaseModel):
    generated_text: str

class StoreDataRequest(BaseModel):
    content: str
    user_id: Optional[str] = None

# -------------------- Endpoints --------------------
@app.get("/")
async def root():
    return {"message": "FastAPI with Gemini and Supabase is running!"}

@app.post("/generate", response_model=GenerateResponse)
async def generate_text(request: GenerateRequest):
    """Generate text using Google Gemini AI."""
    try:
        response = gemini_model.generate_content(
            request.prompt,
            generation_config={"temperature": request.temperature}
        )
        return GenerateResponse(generated_text=response.text)
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Gemini error: {str(e)}")

@app.post("/store")
async def store_data(request: StoreDataRequest):
    """Store a text entry in Supabase (table 'entries')."""
    # You need to create a table named 'entries' with columns: id, content, user_id, created_at
    data = {
        "id": str(uuid.uuid4()),
        "content": request.content,
        "user_id": request.user_id,
    }
    try:
        result = supabase.table("entries").insert(data).execute()
        return {"status": "success", "data": result.data}
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Supabase error: {str(e)}")

@app.post("/upload-file")
async def upload_file(
    file: UploadFile = File(...),
    description: str = Form(...)
):
    """
    Example of handling multipart form data: upload a file and store its info in Supabase.
    """
    # Read file content (for demonstration, we store only metadata)
    file_content = await file.read()
    file_size = len(file_content)

    # Store metadata in Supabase (table 'files')
    data = {
        "id": str(uuid.uuid4()),
        "filename": file.filename,
        "description": description,
        "size_bytes": file_size,
    }
    try:
        result = supabase.table("files").insert(data).execute()
        return {
            "status": "success",
            "filename": file.filename,
            "description": description,
            "size": file_size,
            "supabase_record": result.data
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Supabase error: {str(e)}")
